cmake_minimum_required(VERSION 3.24)
project(QuadSyncAudio VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  add_compile_options(/W4 /utf-8 /permissive-)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_library(quadsync_core INTERFACE)
target_include_directories(quadsync_core INTERFACE native/include)
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE quadsync_core)
add_test(NAME dsp_ring_drift COMMAND core_tests)
if(WIN32)
  add_library(quadsync_engine STATIC native/src/WasapiEngine.cpp)
  target_link_libraries(quadsync_engine PUBLIC quadsync_core ole32 uuid avrt propsys)
  target_compile_definitions(quadsync_engine PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  add_executable(QuadSync.Engine native/src/main.cpp native/src/PipeServer.cpp native/src/Source.cpp)
  target_include_directories(QuadSync.Engine PRIVATE native/third_party)
  target_link_libraries(QuadSync.Engine PRIVATE quadsync_engine advapi32)
  if(MINGW)
    target_link_options(QuadSync.Engine PRIVATE -static)
    target_link_options(core_tests PRIVATE -static)
  endif()
endif()
